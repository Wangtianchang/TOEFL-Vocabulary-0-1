// 在页面加载后，提前加载好语音包 let voices = []; window.speechSynthesis.onvoiceschanged = () => { voices = window.speechSynthesis.getVoices(); }; function playVoice() { const word = document.getElementById('w').innerText; const msg = new SpeechSynthesisUtterance(word); // 核心逻辑：强制查找“最像真人”的高级音库 // 晓晓(Xiaoxiao)和云希(Yunxi)是目前业界公认的最佳发音 const preferredVoice = voices.find(v => v.name.includes('Microsoft Xiaoxiao') || v.name.includes('Microsoft Yunxi') || v.name.includes('Google US English') ); if (preferredVoice) { msg.voice = preferredVoice; } msg.lang = 'en-US'; msg.rate = 0.9; // 稍微放慢，更有学术气息 msg.pitch = 1.0; window.speechSynthesis.cancel(); window.speechSynthesis.speak(msg); }
